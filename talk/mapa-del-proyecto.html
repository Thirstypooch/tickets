@Resolver(() => Event)
export class EventsResolver {
  constructor(
    private eventsService: EventsService,
    private tm: TicketmasterService,
    private reviewsService: ReviewsService,
    private favoritesService: FavoritesService,
  ) {}

  @Query(() => EventDetail, { name: 'eventDetail' })
  async getEventDetail(@Args('id') id: string): Promise<EventDetail> {
    const detail = await this.eventsService.getById(id);

    const [reviews, reviewCount, averageRating] = await Promise.all([
      this.reviewsService.getByEventId(id),
      this.reviewsService.getCountByEventId(id),
      this.reviewsService.getAverageRatingByEventId(id),
    ]);

    const similar = await this.tm.searchEvents({
      category: detail.category, city: detail.city, size: 5,
    });

    return { ...detail, reviews, reviewCount, averageRating,
             similarEvents: similar.events.filter(e => e.id !== id) };
  }
}

final eventRepositoryProvider = Provider<EventRepository>((ref) {
  return ApiEventRepository();
});

final eventFilterProvider = StateProvider<EventFilter>((ref) {
  return const EventFilter();
});

final eventsProvider = FutureProvider<PaginatedEvents>((ref) {
  final filter = ref.watch(eventFilterProvider);           // se re-ejecuta si cambia
  return ref.watch(eventRepositoryProvider).getEvents(filter);
});

final eventDetailProvider =
  FutureProvider.family<EventDetail, String>((ref, id) {
    return ref.watch(eventRepositoryProvider).getEventById(id);
  });

Modelo de datos en Supabase

Cuatro tablas propias. Los eventos no se guardan: viven en Ticketmaster y se referencian por tm_event_id. event_snapshot en bookings congela el evento al momento de comprar (por si Ticketmaster lo modifica después).
